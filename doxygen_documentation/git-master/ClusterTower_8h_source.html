78 size_t generateClusters() {
79 if (getNumActualParticles() == 0) {
80 _clusters.resize(0, Cluster<Particle_T>(nullptr, _clusterSize));
81 _firstOwnedCluster = _clusters.end();
82 _firstTailHaloCluster = _clusters.end();
83 return 0;
84 }
85 this->sortByDim(2);
86
87 // if the number of particles is divisible by the cluster size this is 0
88 const auto sizeLastCluster = this->_particles.size() % _clusterSize;
89 _numDummyParticles = sizeLastCluster == 0 ? 0 : _clusterSize - sizeLastCluster;
90
91 // fill the last cluster with dummy copies of the last particle
92 auto lastParticle = this->_particles[this->_particles.size() - 1];
93 markParticleAsDeleted(lastParticle);
94 for (size_t i = 0; i < _numDummyParticles; i++) {
95 this->addParticle(lastParticle);
96 }
97
98 // Mark start of the different clusters by adding pointers to _particles
99 const size_t numClusters = this->_particles.size() / _clusterSize;
100 _clusters.resize(numClusters, Cluster<Particle_T>(nullptr, _clusterSize));
101 bool foundFirstOwnedCluster = false;
102 bool foundFirstTailHaloCluster = false;
103
104 const auto isAnyOfClusterOwned = [&](const auto &cluster) {
105 for (size_t i = 0; i < _clusterSize; ++i) {
106 if (cluster[i].isOwned()) {
107 return true;
108 }
109 }
110 return false;
111 };
112
113 for (size_t index = 0; index < numClusters; index++) {
114 _clusters[index].reset(&(this->_particles[_clusterSize * index]));
115
116 const bool clusterContainsOwnedParticles =
117 not foundFirstTailHaloCluster and isAnyOfClusterOwned(_clusters[index]);
118
119 // identify the range of owned clusters
120 if (not foundFirstOwnedCluster and clusterContainsOwnedParticles) {
121 _firstOwnedCluster = _clusters.begin() + index;
122 foundFirstOwnedCluster = true;
123 }
124 if (not foundFirstTailHaloCluster and foundFirstOwnedCluster and not clusterContainsOwnedParticles) {
125 _firstTailHaloCluster = _clusters.begin() + index;
126 foundFirstTailHaloCluster = true;
127 }
128 }
129 // for safety, make sure that worst case (e.g. no particles) the iterators point to reasonable positions
130 if (not foundFirstOwnedCluster) {
131 _firstOwnedCluster = _clusters.end();
132 }
133 if (not foundFirstTailHaloCluster) {
134 _firstTailHaloCluster = _clusters.end();
135 }
136
137 return getNumClusters();
138 }

147 void setDummyValues(double dummyStartX, double dummyDistZ) {
148 auto &lastCluster = getCluster(getNumClusters() - 1);
149 for (size_t index = 1; index <= _numDummyParticles; index++) {
150 auto &dummy = lastCluster[_clusterSize - index];
151 dummy = lastCluster[0]; // use first Particle in last cluster as dummy particle!
152 dummy.setOwnershipState(OwnershipState::dummy);
153 dummy.setR({dummyStartX, 0, dummyDistZ * static_cast<double>(index)});
154 dummy.setID(std::numeric_limits<size_t>::max());
155 }

221 std::vector<Particle_T> collectOutOfBoundsParticles(const std::array<double, 3> &boxMin,
222 const std::array<double, 3> &boxMax) {
223 // make sure to get rid of all dummies
224 deleteDummyParticles();
225 // move all particles that are not in the tower to the back of the storage
226 const auto firstOutOfBoundsParticleIter =
227 std::partition(this->_particles.begin(), this->_particles.end(),
228 [&](const auto &p) { return utils::inBox(p.getR(), boxMin, boxMax); });
229
230 // copy out of bounds particles
231 std::vector<Particle_T> outOfBoundsParticles(firstOutOfBoundsParticleIter, this->_particles.end());
232 // shrink the particle storage so all out of bounds particles are cut away
233 this->_particles.resize(std::distance(this->_particles.begin(), firstOutOfBoundsParticleIter),
234 *this->_particles.begin());
235 return outOfBoundsParticles;
236 }

325 void deleteByIndex(size_t index) override {
329
330 // swap particle that should be deleted to end of actual particles.
331 std::swap(this->_particles[index], this->_particles[getNumActualParticles() - 1]);
332 if (getNumTailDummyParticles() != 0) {
333 // swap particle that should be deleted (now at end of actual particles) with last dummy particle.
