41inline std::set<autopas::Configuration> generateAllValidConfigurations(
42 autopas::InteractionTypeOption interactionType,
43 const std::set<autopas::ContainerOption> &allowedContainerOptions = autopas::ContainerOption::getAllOptions(),
44 const std::set<autopas::TraversalOption> &allowedTraversalOptions = autopas::TraversalOption::getAllOptions(),
45 const std::set<autopas::LoadEstimatorOption> &allowedLoadEstimatorOptions =
46 autopas::LoadEstimatorOption::getAllOptions(),
47 const std::set<autopas::DataLayoutOption> &allowedDataLayoutOptions = autopas::DataLayoutOption::getAllOptions(),
48 const std::set<autopas::Newton3Option> &allowedNewton3Options = autopas::Newton3Option::getAllOptions(),
49 const std::set<double> &allowedCellSizeFactors = {0.5, 1.0, 1.5},
50 const std::set<autopas::VectorizationPatternOption> &allowedVectorPatterns =
51 autopas::VectorizationPatternOption::getAllOptions(),
52 bool throwIfNone = true) {
53 const autopas::NumberSetFinite<double> csfs(allowedCellSizeFactors);
54 if (interactionType == autopas::InteractionTypeOption::all) {
55 std::set<autopas::Configuration> allConfigs;
56 for (auto iType : autopas::InteractionTypeOption::getMostOptions()) {
57 // Do not throw for individual interaction types, as only the union has to be non-empty.
58 const auto configs = autopas::SearchSpaceGenerators::cartesianProduct(
59 allowedContainerOptions, allowedTraversalOptions, allowedLoadEstimatorOptions, allowedDataLayoutOptions,
60 allowedNewton3Options, &csfs, allowedVectorPatterns, iType, /*throwIfNone*/ false);
61 allConfigs.insert(configs.begin(), configs.end());
62 }
63 if (throwIfNone and allConfigs.empty()) {
64 autopas::utils::ExceptionHandler::exception("generateAllValidConfigurations: No valid configuration exists.");
65 }
66 return allConfigs;
67 } else {
68 return autopas::SearchSpaceGenerators::cartesianProduct(
69 allowedContainerOptions, allowedTraversalOptions, allowedLoadEstimatorOptions, allowedDataLayoutOptions,
70 allowedNewton3Options, &csfs, allowedVectorPatterns, interactionType, throwIfNone);
71 }

90inline std::optional<autopas::Configuration> getArbitraryConfiguration(
91 std::optional<autopas::InteractionTypeOption> interactionType = std::nullopt,
92 std::optional<autopas::ContainerOption> containerOption = std::nullopt,
93 std::optional<autopas::TraversalOption> traversalOption = std::nullopt,
94 std::optional<autopas::LoadEstimatorOption> loadEstimatorOption = std::nullopt,
95 std::optional<autopas::DataLayoutOption> dataLayoutOption = std::nullopt,
96 std::optional<autopas::Newton3Option> newton3Option = std::nullopt,
97 std::optional<double> cellSizeFactor = std::nullopt,
98 std::optional<autopas::VectorizationPatternOption> vectorPattern = std::nullopt, bool throwIfNone = true) {
99 const auto configurations = generateAllValidConfigurations(
100 interactionType.value_or(autopas::InteractionTypeOption::all),
101 containerOption.has_value() ? std::set<autopas::ContainerOption>{*containerOption}
102 : autopas::ContainerOption::getAllOptions(),
103 traversalOption.has_value() ? std::set<autopas::TraversalOption>{*traversalOption}
104 : autopas::TraversalOption::getAllOptions(),
105 loadEstimatorOption.has_value() ? std::set<autopas::LoadEstimatorOption>{*loadEstimatorOption}
106 : autopas::LoadEstimatorOption::getAllOptions(),
107 dataLayoutOption.has_value() ? std::set<autopas::DataLayoutOption>{*dataLayoutOption}
108 : autopas::DataLayoutOption::getAllOptions(),
109 newton3Option.has_value() ? std::set<autopas::Newton3Option>{*newton3Option}
110 : autopas::Newton3Option::getAllOptions(),
111 cellSizeFactor.has_value() ? std::set<double>{*cellSizeFactor} : std::set<double>{0.5, 1.0, 1.5},
112 vectorPattern.has_value() ? std::set<autopas::VectorizationPatternOption>{*vectorPattern}
113 : autopas::VectorizationPatternOption::getAllOptions(),
114 /*throwIfNone -> false so that we use the error message below which is clearer for users of this function*/
115 false);
116
117 if (configurations.empty()) {
118 if (throwIfNone) {
119 autopas::utils::ExceptionHandler::exception(
120 "getArbitraryConfiguration: No valid configuration exists for the given constraints.");
121 }
122 return std::nullopt;
123 }
124 return *configurations.begin();
125}

219inline std::set<ContainerConfiguration> generateAllValidContainerConfigurations(
220 const std::set<autopas::ContainerOption> &allowedContainerOptions = autopas::ContainerOption::getAllOptions(),
221 const std::set<double> &allowedCellSizeFactors = {0.5, 1.0, 1.5}) {
222 std::set<ContainerConfiguration> containerConfigs;
223 for (const auto &containerOption : allowedContainerOptions) {
224 for (const auto csf : allowedCellSizeFactors) {
225 const ContainerConfiguration containerConfig{containerOption, csf};
226 // Keep this container configuration only if at least one valid full configuration exists for it.
227 if (containerConfig.generateFullConfig(std::nullopt, /*throwIfNone*/ false).has_value()) {
228 containerConfigs.insert(containerConfig);
229 }
230 }
231 }
232 return containerConfigs;
