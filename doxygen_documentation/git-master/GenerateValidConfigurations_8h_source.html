131inline std::set<ContainerConfiguration> generateAllValidContainerConfigurations(
132 const std::set<autopas::ContainerOption> &allowedContainerOptions = autopas::ContainerOption::getAllOptions(),
133 const std::set<double> &allowedCellSizeFactors = {0.5, 1.0, 1.5}) {
134 std::set<ContainerConfiguration> containerConfigs;
135 for (const auto &containerOption : allowedContainerOptions) {
136 for (const auto csf : allowedCellSizeFactors) {
137 // Create a dummy configuration to check validity
138 // We use pairwise interaction as default, since it should not matter for container/csf compatibility.
139 // We also use the first valid traversal for this container.
140 const auto interactionType = autopas::InteractionTypeOption(autopas::InteractionTypeOption::pairwise);
141 const auto traversals = autopas::compatibleTraversals::allCompatibleTraversals(containerOption, interactionType);
142 if (traversals.empty()) {
143 autopas::utils::ExceptionHandler::exception(
144 "{} has no compatible traversals with interaction type {}! This "
145 "suggests that either that something is incorrect with this "
146 "container or that generateAllValidContainerConfigurations's "
147 "assumption that all containers have at least one {} compatible "
148 "traversal no longer holds!",
149 containerOption.to_string(), interactionType.to_string(), interactionType.to_string());
150 continue;
151 }
152 const auto &traversalOption = *traversals.begin();
153 const auto loadEstimators = autopas::loadEstimators::getApplicableLoadEstimators(
154 containerOption, traversalOption, autopas::LoadEstimatorOption::getAllOptions());
155 if (loadEstimators.empty()) {
156 autopas::utils::ExceptionHandler::exception(
157 "{} with traversal {} has no applicable load estimators! Either "
158 "something is incorrect or generateAllValidContainerConfigurations's "
159 "assumption that there is always an applicable load estimator (even "
160 "if 'none') no longer holds.");
161 continue;
162 }
163 const auto &loadEstimatorOption = *loadEstimators.begin();
164
165 const autopas::Configuration configuration{containerOption,
166 csf,
167 traversalOption,
168 loadEstimatorOption,
169 autopas::DataLayoutOption::aos,
170 autopas::Newton3Option::enabled,
171 interactionType};
172 if (configuration.hasCompatibleValues()) {
173 containerConfigs.insert({containerOption, csf});
174 }
175 }
176 }
177 return containerConfigs;
178}
