172 void updateNeighborListsSingleThread(size_t N, double interactionLength, DataLayoutOption dataLayout,
173 bool useNewton3) {
174 std::vector<std::vector<size_t>> tempLists(N);
175
176 typename VerletListHelpers<Particle_T>::CRSNeighborListPolicy policy(tempLists, _particleToIndex);
177
178 InteractionListGeneratorFunctor<Particle_T, typename VerletListHelpers<Particle_T>::CRSNeighborListPolicy> f(
179 policy, interactionLength, useNewton3);
180 auto traversal = LCC08Traversal<
181 ParticleCellType,
182 InteractionListGeneratorFunctor<Particle_T, typename VerletListHelpers<Particle_T>::CRSNeighborListPolicy>>(
183 this->_linkedCells.getCellBlock().getCellsPerDimensionWithHalo(), f, this->getInteractionLength(),
184 this->_linkedCells.getCellBlock().getCellLength(), dataLayout, useNewton3);
185 this->_linkedCells.computeInteractions(&traversal);
186
187 // Prefix-sum + copy:
188 _neighborList.offsets.resize(N + 1);
189 _neighborList.offsets[0] = 0;
190 for (size_t i = 0; i < N; ++i) {
191 _neighborList.offsets[i + 1] = _neighborList.offsets[i] + tempLists[i].size();
192 }
193 const size_t totalNeighbors = _neighborList.offsets[N];
194 _neighborList.indices.resize(totalNeighbors);
195 for (size_t i = 0; i < N; ++i) {
196 std::copy(tempLists[i].begin(), tempLists[i].end(),
197 _neighborList.indices.begin() + static_cast<std::ptrdiff_t>(_neighborList.offsets[i]));
198 }
199
200 AutoPasLog(DEBUG, "VerletLists::updateNeighborLists (1T): {} particles, {} neighbors, avg {:.2f}", N,
201 totalNeighbors, N > 0 ? static_cast<double>(totalNeighbors) / static_cast<double>(N) : 0.0);
202 }

220 void updateNeighborListsMultiThread(size_t N, double interactionLength, DataLayoutOption dataLayout,
221 bool useNewton3) {
222 using PaddedAtomic = VerletListHelpers<Particle_T>::VerletListCounterFunctor::PaddedAtomic;
223
224 // Pass 1: Count neighbors per particle
225 std::vector<PaddedAtomic> counts(N);
226 {
227 typename VerletListHelpers<Particle_T>::VerletListCounterFunctor counter(counts, _particleToIndex,
228 interactionLength);
229 auto traversal =
230 LCC08Traversal<ParticleCellType, typename VerletListHelpers<Particle_T>::VerletListCounterFunctor>(
231 this->_linkedCells.getCellBlock().getCellsPerDimensionWithHalo(), counter, this->getInteractionLength(),
232 this->_linkedCells.getCellBlock().getCellLength(), dataLayout, useNewton3);
233 this->_linkedCells.computeInteractions(&traversal);
234 }
235
236 // Prefix sum: counts -> CRS offsets, allocate flat indices
237 _neighborList.offsets.resize(N + 1);
238 _neighborList.offsets[0] = 0;
239 for (size_t i = 0; i < N; ++i) {
240 _neighborList.offsets[i + 1] = _neighborList.offsets[i] + counts[i].value.load(std::memory_order_relaxed);
241 }
242 const size_t totalNeighbors = _neighborList.offsets[N];
243 _neighborList.indices.resize(totalNeighbors);
244
245 AutoPasLog(DEBUG, "VerletLists::updateNeighborLists (MT): {} particles, {} neighbors, avg {:.2f}", N,
246 totalNeighbors, N > 0 ? static_cast<double>(totalNeighbors) / static_cast<double>(N) : 0.0);
247
248 // Pass 2: Fill CRS indices directly
249 // Reuse the PaddedAtomic array as fill cursors, seeding each with offsets[i].
250 for (size_t i = 0; i < N; ++i) {
251 counts[i].value.store(_neighborList.offsets[i], std::memory_order_relaxed);
252 }
253 {
254 typename VerletListHelpers<Particle_T>::VerletListFillerFunctor filler(_neighborList, counts, _particleToIndex,
255 interactionLength);
256 auto traversal =
257 LCC08Traversal<ParticleCellType, typename VerletListHelpers<Particle_T>::VerletListFillerFunctor>(
258 this->_linkedCells.getCellBlock().getCellsPerDimensionWithHalo(), filler, this->getInteractionLength(),
259 this->_linkedCells.getCellBlock().getCellLength(), dataLayout, useNewton3);
260 this->_linkedCells.computeInteractions(&traversal);
261 }
262 }
