116 BalancedTraversal::EstimatorFunction getLoadEstimatorFunction() {
117 // (Explicit) static cast required for Apple Clang (last tested version: 17.0.0)
118 switch (static_cast<LoadEstimatorOption::Value>(this->_loadEstimator)) {
119 case LoadEstimatorOption::neighborListLength: {
120 return [&](const std::array<unsigned long, 3> &cellsPerDimension,
121 const std::array<unsigned long, 3> &lowerCorner, const std::array<unsigned long, 3> &upperCorner) {
122 // the neighborListLength function defined for verletListsCells in not compatible with this container.
123 unsigned long sum = 0;
124 for (unsigned long x = lowerCorner[0]; x <= upperCorner[0]; x++) {
125 for (unsigned long y = lowerCorner[1]; y <= upperCorner[1]; y++) {
126 unsigned long cellLoad = 0;
127 auto &tower = _towerBlock.getTowerByIndex2D(x, y);
128 for (auto &cluster : tower.getClusters()) {
129 if (cluster.getNeighbors()) {
130 cellLoad += cluster.getNeighbors()->size();
131 }
132 }
133 sum += cellLoad;
134 }
135 }
136 return sum;
137 };
138 }
139 case LoadEstimatorOption::none:
140 [[fallthrough]];
141 default: {
142 return
143 [&](const std::array<unsigned long, 3> &cellsPerDimension, const std::array<unsigned long, 3> &lowerCorner,
144 const std::array<unsigned long, 3> &upperCorner) { return 1; };
145 }
146 }

216 bool updateHaloParticle(const Particle_T &haloParticle) override {
217 using namespace autopas::utils::ArrayMath::literals;
218
219 const auto &haloPos = haloParticle.getR();
220 const auto skinHalf = 0.5 * this->getVerletSkin();
221 // Towers span the whole z range, so only the x-y footprint of the search box matters.
222 const auto [lowX, lowY] = _towerBlock.getTowerIndex2DAtPosition(haloPos - skinHalf);
223 const auto [highX, highY] = _towerBlock.getTowerIndex2DAtPosition(haloPos + skinHalf);
224 for (size_t y = lowY; y <= highY; ++y) {
225 for (size_t x = lowX; x <= highX; ++x) {
226 // It's possible that a tower includes the owned version and a halo version of a particle. But they will be far
227 // enough apart that the position check will ensure no incorrect updates.
228 if (internal::checkParticleInCellAndUpdateByIDAndPosition(_towerBlock.getTowerByIndex2D(x, y), haloParticle,
229 skinHalf)) {
230 return true;
231 }
232 }
233 }
234 AutoPasLog(TRACE, "UpdateHaloParticle was not able to update particle: {}", haloParticle.toString());
235 return false;
236 }

238 void deleteHaloParticles() override {
239 // Step 1: Remove particles from _particlesToAdd
240 for (auto &particleVec : _particlesToAdd) {
241 for (size_t j = 0; j < particleVec.size();) {
242 if (particleVec[j].isHalo()) {
243 particleVec[j] = particleVec[particleVec.size() - 1];
244 particleVec.pop_back();
245 } else {
246 ++j;
247 }
248 }
249 }
250 // Step 2: Remove particles from _towers
251 bool deletedSomething = false;
252 AUTOPAS_OPENMP(parallel for reduction(|| : deletedSomething))
253 // Thanks to clang 13 this has to be a index based loop instead of a range based
254 for (size_t i = 0; i < _towerBlock.size(); ++i) {
255 auto &tower = _towerBlock[i];
256 const auto towerSize = tower.size();
257 auto numTailDummies = tower.getNumTailDummyParticles();
258 // iterate over all non-tail dummies. Avoid underflows.
259 for (size_t j = 0; numTailDummies < towerSize and j < towerSize - numTailDummies;) {
260 if (tower[j].isHalo()) {
261 // swap-"delete"
262 tower[j] = tower[towerSize - 1 - numTailDummies];
263 // Since we can't pop the moved particle here mark it for deletion.
264 internal::markParticleAsDeleted(tower[towerSize - 1 - numTailDummies]);
265 ++numTailDummies;
266 deletedSomething = true;
267 } else {
268 ++j;
269 }
270 }
271 // if anything was marked for deletion actually delete it now.
272 if (deletedSomething) {
273 tower.deleteDummyParticles();
274 }
275 }
276 if (deletedSomething) {
277 _isValid.store(ValidityState::invalid, std::memory_order_relaxed);
278 }
279 }

309 template <bool regionIter>
310 std::tuple<const Particle_T *, size_t, size_t> getParticleImpl(size_t cellIndex, size_t particleIndex,
311 IteratorBehavior iteratorBehavior,
312 const std::array<double, 3> &boxMin,
313 const std::array<double, 3> &boxMax) const {
314 using namespace autopas::utils::ArrayMath::literals;
315
316 // in this context cell == tower
317 // catching the edge case that towers are not yet built -> Iterator jumps to additional vectors
318 if (_towerBlock.empty()) {
319 return {nullptr, 0, 0};
320 }
321
322 std::array<double, 3> boxMinWithSafetyMargin = boxMin;
323 std::array<double, 3> boxMaxWithSafetyMargin = boxMax;
324 if constexpr (regionIter) {
325 // We extend the search box for cells here since particles might have moved
326 boxMinWithSafetyMargin -= 0.5 * this->getVerletSkin();
327 boxMaxWithSafetyMargin += 0.5 * this->getVerletSkin();
328 }
329
330 // first and last relevant cell index
331 const auto [startCellIndex, endCellIndex] = [&]() -> std::tuple<size_t, size_t> {
332 if constexpr (regionIter) {
333 // We extend the search box for cells here since particles might have moved
334 return {_towerBlock.getTowerIndex1DAtPosition(boxMinWithSafetyMargin),
335 _towerBlock.getTowerIndex1DAtPosition(boxMaxWithSafetyMargin)};
336 } else {
337 if (not(iteratorBehavior & IteratorBehavior::halo)) {
338 // only potentially owned region
339 return {_towerBlock.getFirstOwnedTowerIndex(), _towerBlock.getLastOwnedTowerIndex()};
340 } else {
341 // whole range of cells
342 return {0, _towerBlock.size() - 1};
343 }
344 }
345 }();
346
347 // if we are at the start of an iteration ...
348 if (cellIndex == 0 and particleIndex == 0) {
349 cellIndex =
350 startCellIndex + ((iteratorBehavior & IteratorBehavior::forceSequential) ? 0 : autopas_get_thread_num());
351 }
352 // abort if the start index is already out of bounds
353 if (cellIndex >= _towerBlock.size()) {
354 return {nullptr, 0, 0};
355 }
356 // check the data behind the indices
357 if (particleIndex >= _towerBlock[cellIndex].getNumActualParticles() or
358 not containerIteratorUtils::particleFulfillsIteratorRequirements<regionIter>(
359 _towerBlock[cellIndex][particleIndex], iteratorBehavior, boxMin, boxMax)) {
360 // either advance them to something interesting or invalidate them.
361 std::tie(cellIndex, particleIndex) =
362 advanceIteratorIndices<regionIter>(cellIndex, particleIndex, iteratorBehavior, boxMin, boxMax,
363 boxMinWithSafetyMargin, boxMaxWithSafetyMargin, endCellIndex);
364 }
365 // shortcut if the given index doesn't exist
366 if (cellIndex > endCellIndex) {
367 return {nullptr, 0, 0};
368 }
369 const Particle_T *retPtr = &_towerBlock[cellIndex][particleIndex];
370
371 return {retPtr, cellIndex, particleIndex};
372 }

384 [[nodiscard]] std::vector<Particle_T> updateContainer(bool keepNeighborListsValid) override {
385 if (keepNeighborListsValid) {
386 return LeavingParticleCollector::collectParticlesAndMarkNonOwnedAsDummy(*this);
387 }
388 // First delete all halo particles.
389 this->deleteHaloParticles();
390 // Delete dummy particles.
391 AUTOPAS_OPENMP(parallel for)
392 for (size_t i = 0ul; i < _towerBlock.size(); ++i) {
393 _towerBlock[i].deleteDummyParticles();
394 }
395 // Delete dummy particles also from the _particlesToAdd vector
396 AUTOPAS_OPENMP(parallel for)
397 for (size_t i = 0ul; i < _particlesToAdd.size(); ++i) {
398 _particlesToAdd[i].erase(std::remove_if(_particlesToAdd[i].begin(), _particlesToAdd[i].end(),
399 [](const auto &p) { return p.isDummy(); }),
400 _particlesToAdd[i].end());
401 }
402
403 // next find invalid particles
404 std::vector<Particle_T> invalidParticles;
405
406 // custom openmp reduction to concatenate all local vectors to one at the end of a parallel region
407 AUTOPAS_OPENMP(declare reduction(
408 vecMergeParticle : std::vector<Particle_T> : omp_out.insert(omp_out.end(), omp_in.begin(), omp_in.end())))
409 AUTOPAS_OPENMP(parallel reduction(vecMergeParticle : invalidParticles)) {
410 for (auto iter = this->begin(IteratorBehavior::owned); iter.isValid(); ++iter) {
411 if (not utils::inBox(iter->getR(), this->getBoxMin(), this->getBoxMax())) {
412 invalidParticles.push_back(*iter);
413 internal::deleteParticle(iter);
414 }
415 }
416 }
417 _isValid.store(ValidityState::invalid, std::memory_order_relaxed);
418 return invalidParticles;
419 }

1083 void calculateClusterThreadPartition() {
1084 size_t numClusterPairs = 0;
1085 this->template traverseClusters<false>(
1086 [&numClusterPairs](auto &cluster) { numClusterPairs += cluster.getNeighbors()->size(); });
1087
1088 constexpr int minNumClusterPairsPerThread = 1000;
1089 auto numThreads =
1090 std::clamp(static_cast<int>(numClusterPairs / minNumClusterPairsPerThread), 1, autopas_get_max_threads());
1091
1092 size_t numClusterPairsPerThread =
1093 std::max(static_cast<unsigned long>(std::ceil(static_cast<double>(numClusterPairs) / numThreads)), 1ul);
1094 if (numClusterPairsPerThread * numThreads < numClusterPairs) {
1095 autopas::utils::ExceptionHandler::exception(
1096 "VerletClusterLists::calculateClusterThreadPartition(): numClusterPairsPerThread ({}) * numThreads ({})={} "
1097 "should always "
1098 "be at least the amount of Cluster Pairs ({})!",
1099 numClusterPairsPerThread, numThreads, numClusterPairsPerThread * numThreads, numClusterPairs);
1100 }
1101 fillClusterRanges(numClusterPairsPerThread, numThreads);

1103
1110 void fillClusterRanges(size_t numClusterPairsPerThread, int numThreads) {
1111 if (numClusterPairsPerThread < 1) {
1112 autopas::utils::ExceptionHandler::exception(
1113 "VerletClusterLists::fillClusterRanges(): numClusterPairsPerThread({}) is less than one, this is not "
1114 "supported "
1115 "and will lead to errors!",
1116 numClusterPairsPerThread);
1117 }
1118 _clusterThreadPartition.resize(numThreads);
1119
1120 size_t currentThread = 0;
1121 size_t currentNumClustersToAdd = 0;
1122 size_t numClusterPairsTotal = 0;
1123 bool threadIsInitialized = false;
1124 // Iterate over the clusters of all towers
1125 for (size_t currentTowerIndex = 0; currentTowerIndex < _towerBlock.size(); currentTowerIndex++) {
1126 auto &currentTower = _towerBlock[currentTowerIndex];
1127 const auto firstOwnedClusterIndex =
1128 std::distance(currentTower.getClusters().begin(), currentTower.getFirstOwnedCluster());
1129 const auto firstTailHaloClusterIndex =
1130 std::distance(currentTower.getClusters().begin(), currentTower.getFirstTailHaloCluster());
1131 for (size_t currentClusterInTower = firstOwnedClusterIndex; currentClusterInTower < firstTailHaloClusterIndex;
1132 ++currentClusterInTower) {
1133 auto &currentCluster = currentTower.getCluster(currentClusterInTower);
1134
1135 // If on a new thread, start with the clusters for this thread here.
1136 if (not threadIsInitialized) {
1137 _clusterThreadPartition[currentThread] = {currentTowerIndex, currentClusterInTower, 0};
1138 threadIsInitialized = true;
1139 }
1140
1141 currentNumClustersToAdd++;
1142 numClusterPairsTotal += currentCluster.getNeighbors()->size();
1143
1144 // If the thread is finished, write number of clusters and start new thread.
1145 if (numClusterPairsTotal >= numClusterPairsPerThread * (currentThread + 1)) {
1146 // Add the number of clusters for the finished thread.
1147 _clusterThreadPartition[currentThread].numClusters += currentNumClustersToAdd;
1148 currentNumClustersToAdd = 0;
1149 // Go to next thread!
1150 currentThread++;
1151 // if we are already at the end of all threads, go back to last thread!
1152 // this is a safety precaution and should not really matter.
1153 if (currentThread >= numThreads) {
1154 --currentThread;
1155 threadIsInitialized = true;
1156 } else {
1157 threadIsInitialized = false;
1158 }
1159 }
1160 }
1161 }
1162 if (not threadIsInitialized) {
1163 _clusterThreadPartition[currentThread] = {0, 0, 0};
1164 }
1165 // Make sure the last cluster range contains the rest of the clusters, even if there is not the perfect number left.
1166 if (currentNumClustersToAdd != 0) {
1167 _clusterThreadPartition[currentThread].numClusters += currentNumClustersToAdd;
1168 }
1169 // Theoretically, some threads may still remain. This ensures that their numClusters are set to 0.
1170 while (++currentThread < numThreads) {
1171 _clusterThreadPartition[currentThread] = {0, 0, 0};
1172 }
1173 }

1200 template <bool regionIter>
1201 [[nodiscard]] std::tuple<size_t, size_t> advanceIteratorIndices(
1202 size_t cellIndex, size_t particleIndex, IteratorBehavior iteratorBehavior, const std::array<double, 3> &boxMin,
1203 const std::array<double, 3> &boxMax, const std::array<double, 3> &boxMinWithSafetyMargin,
1204 const std::array<double, 3> &boxMaxWithSafetyMargin, size_t endCellIndex) const {
1205 // Finding the indices for the next particle
1206 const size_t stride = (iteratorBehavior & IteratorBehavior::forceSequential) ? 1 : autopas_get_num_threads();
1207
1208 // helper function to determine if the cell can even contain particles of interest to the iterator
1209 auto towerIsRelevant = [&]() -> bool {
1210 // special case: Towers are not yet built, then the tower acting as buffer is always relevant.
1211 if (_towerBlock.size() == 1) {
1212 return true;
1213 }
1214 bool isRelevant = true;
1215 if constexpr (regionIter) {
1216 // is the cell in the region?
1217 const auto [towerLowCorner, towerHighCorner] = _towerBlock.getTowerBoundingBox(cellIndex);
1218 isRelevant =
1219 utils::boxesOverlap(towerLowCorner, towerHighCorner, boxMinWithSafetyMargin, boxMaxWithSafetyMargin);
1220 }
1221 return isRelevant;
1222 };
1223
1224 do {
1225 // advance to the next particle
1226 ++particleIndex;
1227 // If this breaches the end of a cell, find the next non-empty cell and reset particleIndex.
1228
1229 // If cell has wrong type, or there are no more particles in this cell jump to the next
1230 while (not towerIsRelevant() or particleIndex >= _towerBlock[cellIndex].getNumActualParticles()) {
1231 cellIndex += stride;
1232 particleIndex = 0;
1233
1234 // If we notice that there is nothing else to look at set invalid values, so we get a nullptr next time and
1235 // break.
1236 if (cellIndex > endCellIndex) {
1237 return {std::numeric_limits<size_t>::max(), particleIndex};
1238 }
1239 }
1240 } while (not containerIteratorUtils::particleFulfillsIteratorRequirements<regionIter>(
1241 _towerBlock[cellIndex][particleIndex], iteratorBehavior, boxMin, boxMax));
1242
1243 // the indices returned at this point should always be valid
1244 return {cellIndex, particleIndex};
1245 }
