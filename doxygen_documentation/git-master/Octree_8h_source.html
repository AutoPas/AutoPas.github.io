122 [[nodiscard]] std::vector<ParticleType> updateContainer(bool keepNeighborListValid) override {
123 // invalidParticles: all outside boxMin/Max
124 std::vector<Particle_T> invalidParticles{};
125
126 if (keepNeighborListValid) {
127 invalidParticles = LeavingParticleCollector::collectParticlesAndMarkNonOwnedAsDummy(*this);
128 } else {
129 // This is a very primitive and inefficient way to rebuild the container:
130
131 // @todo Make this less indirect. (Find a better way to iterate all particles inside the octree to change
132 // this function back to a function that actually copies all particles out of the octree.)
133 // The problem is captured by https://github.com/AutoPas/AutoPas/issues/622
134
135 // 1. Copy all particles out of the container
136 std::vector<Particle_T *> particleRefs;
137 this->_cells[CellTypes::OWNED].collectAllParticles(particleRefs);
138 for (auto &buffer : _particlesToAdd[CellTypes::OWNED]) {
139 for (auto &particle : buffer._particles) {
140 particleRefs.push_back(&particle);
141 }
142 }
143 std::vector<Particle_T> particles{};
144 particles.reserve(particleRefs.size());
145
146 for (auto *p : particleRefs) {
147 if (p->isDummy()) {
148 // don't do anything with dummies. They will just be dropped when the container is rebuilt.
149 continue;
150 } else if (utils::inBox(p->getR(), this->getBoxMin(), this->getBoxMax())) {
151 particles.push_back(*p);
152 } else {
153 invalidParticles.push_back(*p);
154 }
155 }
156
157 // 2. Clear the container
158 this->deleteAllParticles();
159
160 // 3. Insert the particles back into the container
161 for (auto &particle : particles) {
162 this->_cells[CellTypes::OWNED].addParticle(particle);
163 }
164 }
165
166 return invalidParticles;
167 }

280 template <bool regionIter>
281 std::tuple<const ParticleType *, size_t, size_t> getParticleImpl(size_t cellIndex, size_t particleIndex,
282 IteratorBehavior iteratorBehavior,
283 const std::array<double, 3> &boxMin,
284 const std::array<double, 3> &boxMax) const {
285 using namespace autopas::utils::ArrayMath::literals;
286 // FIXME think about parallelism.
287 // This `if` currently disables it but should be replaced with logic that determines the start index.
288 if (autopas_get_thread_num() > 0 and not(iteratorBehavior & IteratorBehavior::forceSequential)) {
289 return {nullptr, 0, 0};
290 }
291 // if owned particles are not interesting jump directly to the halo tree
292 // FIXME: with num threads > 1 the start cell IDs will have to be set to more complicated values here
293 if (cellIndex == 0 and not(iteratorBehavior & IteratorBehavior::owned)) {
294 cellIndex = HALO;
295 }
296
297 // shortcut if the given index doesn't exist
298 if (cellIndex < 10 and cellIndex > HALO) {
299 return {nullptr, 0, 0};
300 }
301
302 std::array<double, 3> boxMinWithSafetyMargin = boxMin;
303 std::array<double, 3> boxMaxWithSafetyMargin = boxMax;
304 if constexpr (regionIter) {
305 // We extend the search box for cells here since particles might have moved
306 boxMinWithSafetyMargin -= 0.5 * this->getVerletSkin();
307 boxMaxWithSafetyMargin += 0.5 * this->getVerletSkin();
308 }
309
310 std::vector<size_t> currentCellIndex{};
311 OctreeLeafNode<Particle_T> *currentCellPtr = nullptr;
312
313 std::tie(currentCellIndex, currentCellPtr) = getLeafCellByIndex(cellIndex);
314 // check the data behind the indices
315 if (particleIndex >= currentCellPtr->size() or
316 not containerIteratorUtils::particleFulfillsIteratorRequirements<regionIter>(
317 (*currentCellPtr)[particleIndex], iteratorBehavior, boxMin, boxMax)) {
318 // either advance them to something interesting or invalidate them.
319 std::tie(currentCellPtr, particleIndex) =
320 advanceIteratorIndices<regionIter>(currentCellIndex, currentCellPtr, particleIndex, iteratorBehavior, boxMin,
321 boxMax, boxMinWithSafetyMargin, boxMaxWithSafetyMargin);
322 }
323
324 // shortcut if the given index doesn't exist
325 if (currentCellPtr == nullptr) {
326 return {nullptr, 0, 0};
327 }
328 // parse cellIndex and get referenced cell and particle
329 const Particle_T *retPtr = &((*currentCellPtr)[particleIndex]);
330
331 // if no err value was set, convert cell index from vec to integer
332 if (currentCellIndex.empty()) {
333 cellIndex = invalidCellIndex;
334 } else {
335 cellIndex = 0;
336 // needs signed int to prevent underflow
337 for (int i = static_cast<int>(currentCellIndex.size()) - 1; i >= 0; --i) {
338 cellIndex *= 10;
339 cellIndex += currentCellIndex[i];
340 }
341 }
342
343 return {retPtr, cellIndex, particleIndex};
344 }

355 std::tuple<std::vector<size_t>, OctreeLeafNode<Particle_T> *> getLeafCellByIndex(size_t cellIndex) const {
356 // parse cellIndex and get referenced cell and particle
357 std::vector<size_t> currentCellIndex;
358 // constant heuristic for the tree depth
359 currentCellIndex.reserve(10);
360 currentCellIndex.push_back(cellIndex % 10);
361 cellIndex /= 10;
362 OctreeNodeInterface<Particle_T> *currentCell = this->_cells[currentCellIndex.back()].getRaw();
363 // don't restrict loop via cellIndex because it might have "hidden" leading 0
364 while (currentCell->hasChildren()) {
365 currentCellIndex.push_back(cellIndex % 10);
366 cellIndex /= 10;
367 currentCell = currentCell->getChild(currentCellIndex.back());
368 }
369 return {currentCellIndex, dynamic_cast<OctreeLeafNode<Particle_T> *>(currentCell)};
370 }

594 template <bool regionIter>
595 std::tuple<OctreeLeafNode<Particle_T> *, size_t> advanceIteratorIndices(
596 std::vector<size_t> &currentCellIndex, OctreeNodeInterface<Particle_T> *const currentCellPtr,
597 size_t particleIndex, IteratorBehavior iteratorBehavior, const std::array<double, 3> &boxMin,
598 const std::array<double, 3> &boxMax, const std::array<double, 3> &boxMinWithSafetyMargin,
599 const std::array<double, 3> &boxMaxWithSafetyMargin) const {
600 // TODO: parallelize at the higher tree levels. Choose tree level to parallelize via log_8(numThreads)
601 const size_t minLevel = 0;
602 // (iteratorBehavior & IteratorBehavior::forceSequential) or autopas_get_num_threads() == 1
603 // ? 0
604 // : static_cast<size_t>(std::ceil(std::log(static_cast<double>(autopas_get_num_threads())) /
605 // std::log(8.)));
606 OctreeNodeInterface<Particle_T> *currentCellInterfacePtr = currentCellPtr;
607 OctreeLeafNode<Particle_T> *currentLeafCellPtr = nullptr;
608
609 // helper function:
610 auto cellIsRelevant = [&](const OctreeNodeInterface<Particle_T> *const cellPtr) {
611 bool isRelevant = cellPtr->size() > 0;
612 if constexpr (regionIter) {
613 isRelevant = utils::boxesOverlap(cellPtr->getBoxMin(), cellPtr->getBoxMax(), boxMinWithSafetyMargin,
614 boxMaxWithSafetyMargin);
615 }
616 return isRelevant;
617 };
618
619 // find the next particle of interest which might be in a different cell
620 do {
621 // advance to the next particle
622 ++particleIndex;
623
624 // this loop finds the next relevant leaf cell or triggers a return
625 // flag for weird corner cases. See further down.
626 bool forceJumpToNextCell = false;
627 // If this breaches the end of a cell, find the next non-empty cell and reset particleIndex.
628 while (particleIndex >= currentCellInterfacePtr->size() or forceJumpToNextCell) {
629 // CASE: we are at the end of a branch
630 // => Move up until there are siblings that were not touched yet
631 while (currentCellIndex.back() == 7) {
632 currentCellInterfacePtr = currentCellInterfacePtr->getParent();
633 currentCellIndex.pop_back();
634 // If there are no more cells in the branch that we are responsible for set invalid parameters and return.
635 if (currentCellIndex.size() < minLevel or currentCellIndex.empty()) {
636 currentCellIndex.clear();
637 return {nullptr, std::numeric_limits<decltype(particleIndex)>::max()};
638 }
639 }
640
641 // Switch to the next child of the same parent
642 ++currentCellIndex.back();
643 // Identify the next (inner) cell pointer
644 if (currentCellIndex.size() == 1) {
645 // if we are already beyond everything
646 if (currentCellIndex.back() > HALO) {
647 return {nullptr, std::numeric_limits<decltype(particleIndex)>::max()};
648 }
649 // special case: the current thread should ALSO iterate halo particles
650 if ((iteratorBehavior & IteratorBehavior::halo)
651 /* FIXME: for parallelization: and ((iteratorBehavior & IteratorBehavior::forceSequential) or autopas_get_num_threads() == 1) */) {
652 currentCellInterfacePtr = this->_cells[HALO].getRaw();
653 } else {
654 // don't jump to the halo tree -> set invalid parameters and return
655 currentCellIndex.clear();
656 return {nullptr, std::numeric_limits<decltype(particleIndex)>::max()};
657 }
658 } else {
659 currentCellInterfacePtr = currentCellInterfacePtr->getParent()->getChild(currentCellIndex.back());
660 }
661 // check that the inner cell is actually interesting, otherwise skip it.
662 if (not cellIsRelevant(currentCellInterfacePtr)) {
663 forceJumpToNextCell = true;
664 continue;
665 }
666 // The inner cell is relevant so descend to its first relevant leaf
667 forceJumpToNextCell = false;
668 while (currentCellInterfacePtr->hasChildren() and not forceJumpToNextCell) {
669 // find the first child in the relevant region
670 size_t firstRelevantChild = 0;
671 // if the child is irrelevant (empty or outside the region) skip it
672 const auto *childToCheck = currentCellInterfacePtr->getChild(firstRelevantChild);
673 while (not cellIsRelevant(childToCheck)) {
674 ++firstRelevantChild;
675 childToCheck = currentCellInterfacePtr->getChild(firstRelevantChild);
676 if (firstRelevantChild > 7) {
677 // weird corner case: we descended into this branch because it overlaps with the region and has particles
678 // BUT the particles are not inside the children which are in the region,
679 // hence no child fulfills both requirements and all are irrelevant.
680 forceJumpToNextCell = true;
681 break;
682 }
683 }
684 currentCellIndex.push_back(firstRelevantChild);
685 currentCellInterfacePtr = currentCellInterfacePtr->getChild(firstRelevantChild);
686 }
687 particleIndex = 0;
688 }
689
690 // at this point we should point to a leaf. All other cases should have hit a return earlier.
691 currentLeafCellPtr = dynamic_cast<OctreeLeafNode<Particle_T> *>(currentCellInterfacePtr);
692 // sanity check
693 if (currentLeafCellPtr == nullptr) {
694 utils::ExceptionHandler::exception("Expected a leaf node but didn't get one!");
695 }
696
697 } while (not containerIteratorUtils::particleFulfillsIteratorRequirements<regionIter>(
698 (*currentLeafCellPtr)[particleIndex], iteratorBehavior, boxMin, boxMax));
699 return {currentLeafCellPtr, particleIndex};
700 }

715 template <class ParticleVec_T, class ParticlesToAdd_T>
716 static ParticleVec_T gatherAdditionalVectors(utils::optRef<ParticleVec_T> additionalVectors,
717 ParticlesToAdd_T &particlesToAdd) {
718 ParticleVec_T allAdditionalVectors;
719 if (additionalVectors.has_value()) {
720 allAdditionalVectors = additionalVectors->get();
721 }
722 for (auto &buffers : particlesToAdd) {
