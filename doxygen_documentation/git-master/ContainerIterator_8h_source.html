59template <bool regionIter, class Particle_T, class Arr>
60[[nodiscard]] bool particleFulfillsIteratorRequirements(const Particle_T &p, IteratorBehavior behavior,
61 const Arr &regionMin, const Arr &regionMax) {
62 // If this is a region iterator check box condition first, otherwise race conditions can occur
63 // if multiple region iterator overlap and ownership state is touched.
64 // (e.g. during collection of leaving particles)
65 if constexpr (regionIter) {
66 if (not utils::inBox(p.getR(), regionMin, regionMax)) {
67 return false;
68 }
69 }
70
71 // Check ownership
72 // Dummy is not in sync with iterator behavior because it needs to be 0.
73 // `a & b == b` idiom is to check a == b disregarding any bits beyond b. This is needed due to e.g. forceSequential.
74 if (((behavior & IteratorBehavior::ownedOrHaloOrDummy) == IteratorBehavior::ownedOrHaloOrDummy) or
75 (behavior & IteratorBehavior::dummy and p.isDummy())) {
76 return true;
77 } else {
78 // relies on both enums having the same encoding. This should be guaranteed statically in IteratorBehaviorTest!
79 return static_cast<unsigned int>(p.getOwnershipState()) & static_cast<unsigned int>(behavior);
80 }
81}

325 void fetchParticleAtCurrentIndex() {
326 if (not _iteratingAdditionalVectors) {
327 // getParticle either gives us a particle with the desired properties or a nullptr
328 if constexpr (regionIter) {
329 std::tie(_currentParticle, _currentVectorIndex, _currentParticleIndex) =
330 _container->getParticle(_currentVectorIndex, _currentParticleIndex, _behavior, _regionMin, _regionMax);
331 } else {
332 std::tie(_currentParticle, _currentVectorIndex, _currentParticleIndex) =
333 _container->getParticle(_currentVectorIndex, _currentParticleIndex, _behavior);
334 }
335 // if getParticle told us that the container doesn't have a particle in the first vector for our thread...
336 if (_currentParticle == nullptr and not _additionalVectors.empty()) {
337 // determine which additional vector this thread should start to iterate
338 _currentVectorIndex = (_behavior & IteratorBehavior::forceSequential) ? 0 : autopas_get_thread_num();
339 _currentParticleIndex = 0;
340 _iteratingAdditionalVectors = true;
341 } else {
342 // Case: nothing left in the container and no additional vectors to iterate
343 return;
344 }
345 }
346 // no "else" here because the first case might trigger the second if we reached the end of the container
347 if (_iteratingAdditionalVectors) {
348 // check all vectors ...
349 for (; _currentVectorIndex < _additionalVectors.size(); _currentVectorIndex += _vectorIndexOffset) {
350 // ... and all particles within the vector ...
351 for (; _currentParticleIndex < _additionalVectors[_currentVectorIndex]->size(); ++_currentParticleIndex) {
352 _currentParticle = &(_additionalVectors[_currentVectorIndex]->operator[](_currentParticleIndex));
353 // ... until we find a particle that satisfies our requirements.
354 if (containerIteratorUtils::particleFulfillsIteratorRequirements<regionIter>(*_currentParticle, _behavior,
355 _regionMin, _regionMax)) {
356 return;
357 }
358 }
359 _currentParticleIndex = 0;
360 }
361 }
362 // if we reach this point there is no satisfying particle left neither in the container nor the additional vectors.
363 _currentParticle = nullptr;
364 _currentParticleIndex = std::numeric_limits<decltype(_currentParticleIndex)>::max();
365 _currentVectorIndex = std::numeric_limits<decltype(_currentVectorIndex)>::max();
366 }
367
374 void deleteCurrentParticle() {
375 if (_iteratingAdditionalVectors) {
376 internal::markParticleAsDeleted(*_currentParticle);
377 } else {
378 _container->deleteParticle(_currentVectorIndex, _currentParticleIndex);
379 }
380 // move on to the next particle, as the deleted one stays in place.
381 this->operator++();
382 // move the current index back so that the next ++ points again to the same particle.
383 --_currentParticleIndex;
384 }
