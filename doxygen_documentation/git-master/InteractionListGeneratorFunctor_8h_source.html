99 void AoSFunctor(Particle_T &i, Particle_T &j, bool newton3) override {
100 using namespace autopas::utils::ArrayMath::literals;
101
102 if (_gatherNewton3Lists and not newton3) [[unlikely]] {
103 utils::ExceptionHandler::exception(
104 "InteractionListGeneratorFunctor should not be used with newton3=false and gatherNewton3Lists=true.");
105 }
106
107 if (i.isDummy() or j.isDummy()) {
108 return;
109 }
110 auto displacement = i.getR() - j.getR();
111
112 double distSquared = utils::ArrayMath::dot(displacement, displacement);
113 if (distSquared < _interactionLengthSquared) {
114 // Assuming this functor is used like any other functor, this is generally thread safe due to how AutoPas
115 // traversals are designed. However, the neighbor list policy and the underlying data structure must be suitable
116 // as well. (e.g. no unwanted resizing during the traversal)
117
118 // - If newton3=false & gatherNewton3Lists=false, we only need to add the i->j interaction to i's list as the j->i
119 // interaction is handled in another call.
120 // - If newton3=true & gatherNewton3Lists=false, we need to add the i->j interaction to j's list and the j->i
121 // interaction to j's list.
122 // - If newton3=true & gatherNewton3Lists=true, we only need to add the i->j interaction to i's list as we don't
123 // want the j->i interaction in the lists. (Could also be the other way around)
124
125 _neighborListPolicy.add(&i, &j);
126 if (newton3 and not _gatherNewton3Lists) {
127 _neighborListPolicy.add(&j, &i);
128 }
129 }
130 }

136 void SoAFunctorSingle(SoAView<SoAArraysType> soa, bool /*newton3*/) override {
137 if (soa.size() == 0) return;
138
139 auto **const __restrict ptrptr = soa.template begin<Particle_T::AttributeNames::ptr>();
140 const double *const __restrict xptr = soa.template begin<Particle_T::AttributeNames::posX>();
141 const double *const __restrict yptr = soa.template begin<Particle_T::AttributeNames::posY>();
142 const double *const __restrict zptr = soa.template begin<Particle_T::AttributeNames::posZ>();
143 const auto *const __restrict ownedStatePtr = soa.template begin<Particle_T::AttributeNames::ownershipState>();
144
145 size_t numPart = soa.size();
146 for (size_t i = 0; i < numPart; ++i) {
147 if (ownedStatePtr[i] == OwnershipState::dummy) continue;
148
149 for (size_t j = i + 1; j < numPart; ++j) {
150 if (ownedStatePtr[j] == OwnershipState::dummy) continue;
151 const double drx = xptr[i] - xptr[j];
152 const double dry = yptr[i] - yptr[j];
153 const double drz = zptr[i] - zptr[j];
154
155 const double drx2 = drx * drx;
156 const double dry2 = dry * dry;
157 const double drz2 = drz * drz;
158
159 const double distSquared = drx2 + dry2 + drz2;
160
161 if (distSquared < _interactionLengthSquared) {
162 // This SoAFunctorSingle implementation always used newton3 in practice, regardless of the option.
163 // This is thread safe (see comment in AoS functor)
164 _neighborListPolicy.add(ptrptr[i], ptrptr[j]);
165 if (not _gatherNewton3Lists) {
166 _neighborListPolicy.add(ptrptr[j], ptrptr[i]);
167 }
168 }
169 }
