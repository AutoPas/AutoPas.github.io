155 static std::shared_ptr<spdlog::logger> create_internal(std::ostream &oss) {
156 std::shared_ptr<spdlog::sinks::sink> ostreamSink;
157#ifdef AUTOPAS_COLORED_CONSOLE_LOGGING
158 if (oss.rdbuf() == std::cout.rdbuf()) {
159 ostreamSink = std::make_shared<spdlog::sinks::stdout_color_sink_mt>();
160 } else if (oss.rdbuf() == std::cerr.rdbuf()) {
161 ostreamSink = std::make_shared<spdlog::sinks::stderr_color_sink_mt>();
162 } else { // no color for streams other than cout / cerr
163 ostreamSink = std::make_shared<spdlog::sinks::ostream_sink_mt>(oss);
164 }
165#else
166 ostreamSink = std::make_shared<spdlog::sinks::ostream_sink_mt>(oss);
167#endif
168 auto logger = std::make_shared<spdlog::logger>(loggerName, ostreamSink);
169 // The logger is normally only flushed on successful program termination.
170 // This line ensures flushing when log messages of level warning or more severe are created.
171 logger->flush_on(spdlog::level::warn);
172 spdlog::register_logger(logger);
173 return logger;
174 }
