22 const std::array<double, 3> &boxMax) {
23 for (size_t d = 0; d < 3; ++d) {
24 const double boxLength = boxMax[d] - boxMin[d];
25 if (pos[d] < boxMin[d]) {
26 pos[d] += boxLength;

44template <class Particle_T>
45std::vector<Particle_T> generatePeriodicImagesOfSingleParticle(const Particle_T &particle,
46 const std::array<double, 3> &boxMin,
47 const std::array<double, 3> &boxMax,
48 const double interactionLength) {
49 std::vector<Particle_T> images;
50 const auto &pos = particle.getR();
51 // Check every face, edge, and corner of the box, given as a direction from the box center. The particle has an
52 // image on the opposite side if it is within the interaction length of all boundaries of this direction.
53 for (const int x : {-1, 0, 1}) {
54 for (const int y : {-1, 0, 1}) {
55 for (const int z : {-1, 0, 1}) {
56 if (x == 0 and y == 0 and z == 0) {
57 continue;
58 }
59 const std::array<int, 3> direction{x, y, z};
60 bool nearBoundaries = true;
61 std::array<double, 3> shift{};
62 for (size_t d = 0; d < 3; ++d) {
63 const double boxLength = boxMax[d] - boxMin[d];
64 if (direction[d] == -1) {
65 nearBoundaries = nearBoundaries and pos[d] < boxMin[d] + interactionLength;
66 shift[d] = boxLength;
67 } else if (direction[d] == 1) {
68 nearBoundaries = nearBoundaries and pos[d] >= boxMax[d] - interactionLength;
69 shift[d] = -boxLength;
70 }
71 }
72 if (nearBoundaries) {
73 auto image = particle;
74 image.addR(shift);
75 images.push_back(image);
76 }
77 }
78 }
79 }
80 return images;
81}
