60 void addParticleReference(size_t index) {
61 if (_particleStorage == nullptr) {
62 autopas::utils::ExceptionHandler::exception(
63 "ReferenceParticleCell::addParticleReference() called before setParticleStorage().");
64 }
65 const Particle_T &p = (*_particleStorage)[index];
66 // sanity check that ensures that only particles of the cells OwnershipState can be added. Note: if a cell is a
67 // dummy-cell, only dummies can be added, otherwise dummies can always be added
68 if ((not toInt64(p.getOwnershipState() & this->_ownershipState)) and
69 p.getOwnershipState() != OwnershipState::dummy) {
70 autopas::utils::ExceptionHandler::exception(
71 "ReferenceParticleCell::addParticleReference() can not add a particle with OwnershipState {} to a cell with "
72 "OwnershipState {}",
73 p.getOwnershipState(), this->_ownershipState);
74 }

77 _particles.push_back(index);
