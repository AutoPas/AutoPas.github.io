37inline size_t getNumberOfParticles(const std::array<double, 3> &boxMin, const std::array<double, 3> &boxMax,
38 const double spacing = 1.0, const bool centeredAlignment = false) {
39 if (spacing <= 0.0) {
40 return 0;
41 }
42
43 const double latticeConstant = std::sqrt(2.0) * spacing;
44 std::array<double, 3> offset = {0.0, 0.0, 0.0};
45 if (centeredAlignment) {
46 offset = {latticeConstant / 4.0, latticeConstant / 4.0, latticeConstant / 4.0};
47 }
48
49 auto countIndices = [&](const double bMin, const double bMax, const double offsetVal,
50 const double basisVal) -> size_t {
51 const double startPos = bMin + offsetVal + basisVal * latticeConstant;
52 if (startPos >= bMax) {
53 return 0;
54 }
55 const double remaining = bMax - startPos;
56 const auto count = static_cast<size_t>(std::ceil(remaining / latticeConstant));
57 return count;
58 };
59
60 size_t totalCount = 0;
61 for (const auto &basis : fccBasis) {
62 size_t basisCount = 1;
63 for (size_t d = 0; d < 3; ++d) {
64 basisCount *= countIndices(boxMin[d], boxMax[d], offset[d], basis[d]);
65 }
66 totalCount += basisCount;
67 }
68 return totalCount;
