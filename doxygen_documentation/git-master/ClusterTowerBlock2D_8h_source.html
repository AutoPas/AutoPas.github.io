131 [[nodiscard]] std::tuple<std::array<double, 2>, std::array<size_t, 2>> estimateOptimalGridSideLength(
132 size_t numParticles, size_t clusterSize) const {
133 using namespace autopas::utils::ArrayMath::literals;
134 using utils::ArrayMath::ceil;
135 using utils::ArrayUtils::static_cast_copy_array;
136
137 // estimate the grid size for the actual box, then add halo layers as necessary
138 const std::array<double, 3> boxSize = _boxMax - _boxMin;
139 const std::array<double, 2> boxSize2D{boxSize[0], boxSize[1]};
140 if (numParticles == 0) {
141 // One owned tower per direction plus the halo towers filling one interaction length on each side, so at least 3
142 // towers per direction. This is needed for sliced traversals to work correctly.
143 // As AutoPas supports only box length >= cutoff, the calculation below is overkill. But I guess maybe useful in
144 // the future?
145 const auto numTowers = ceil(_interactionLength / boxSize2D) * 2. + 1.;
146 return {boxSize2D, static_cast_copy_array<size_t>(numTowers)};
147 }
148
149 const double volume = boxSize[0] * boxSize[1] * boxSize[2];
150 // We use double here to avoid back and forth casting
151 // estimate particle density
152 const double density = static_cast<double>(numParticles) / volume;
153 const auto optimalSideLength = std::cbrt(static_cast<double>(clusterSize) / density);
154 // make sure the towers fill the domain exactly
155 const auto numTowersOwned(ceil(boxSize2D / optimalSideLength));
156 const auto towerSideLengthNew = boxSize2D / numTowersOwned;
157 // pad a halo layer of towers in both directions, filling one interaction length each.
158 const auto numTowers = numTowersOwned + (ceil(_interactionLength / towerSideLengthNew) * 2.);
159 // Do not resize the number of towers here!
160 // If we get less towers we need to save the particles in the towers we throw away.
161 return {towerSideLengthNew, static_cast_copy_array<size_t>(numTowers)};
162 }

186 std::tuple<std::array<double, 3>, std::array<double, 3>> getTowerBoundingBox(
187 const std::array<size_t, 2> &index2D) const {
188 // case: towers are not built yet.
189 if (_towersPerDim[0] == 0) {
190 return {_boxMin, _boxMax};
191 }
192 // towerBoxMin[0/1] does NOT start at _haloBoxMin[0/1] because the halo towers might extend beyond the halo box.
193 const std::array<double, 3> towerBoxMin{
194 _boxMin[0] - _towerSideLength[0] * _numTowersPerInteractionLength2D[0] +
195 _towerSideLength[0] * static_cast<double>(index2D[0]),
196 _boxMin[1] - _towerSideLength[1] * _numTowersPerInteractionLength2D[1] +
197 _towerSideLength[1] * static_cast<double>(index2D[1]),
198 _haloBoxMin[2],
199 };
200 const std::array<double, 3> towerBoxMax{
201 towerBoxMin[0] + _towerSideLength[0],
202 towerBoxMin[1] + _towerSideLength[1],
203 _haloBoxMax[2],
204 };
205 return {towerBoxMin, towerBoxMax};
206 }

214 [[nodiscard]] std::array<size_t, 2> getTowerIndex2DAtPosition(const std::array<double, 3> &pos) const {
215 // special case: Towers are not yet built, then everything is in this one tower.
216 if (_towers.size() == 1) {
217 return {0, 0};
218 }
219
220 std::array<size_t, 2> towerIndex2D{};
221
222 for (int dim = 0; dim < 2; dim++) {
223 const auto towerDimIndex =
224 static_cast<long int>(std::floor((pos[dim] - _boxMin[dim]) * _towerSideLengthReciprocal[dim])) +
225 _numTowersPerInteractionLength2D[dim];
226 // Rounding can put the index of a position next to the box boundary on the wrong side of it.
227 const long int numHaloTowers = _numTowersPerInteractionLength2D[dim];
228 const auto numTowers = static_cast<long int>(_towersPerDim[dim]);
229 long int minIndex = numHaloTowers;
230 long int maxIndex = numTowers - numHaloTowers - 1;
231 if (pos[dim] < _boxMin[dim]) {
232 minIndex = 0;
233 maxIndex = numHaloTowers - 1;
234 } else if (pos[dim] >= _boxMax[dim]) {
235 minIndex = numTowers - numHaloTowers;
236 maxIndex = numTowers - 1;
237 }
238 towerIndex2D[dim] = static_cast<size_t>(std::clamp(towerDimIndex, minIndex, maxIndex));
240 // flag manager
241 if (pos[dim] >= _haloBoxMax[dim]) {
242 towerIndex2D[dim] = _towersPerDim[dim] - 1;
243 } else if (pos[dim] < _haloBoxMin[dim]) {
244 towerIndex2D[dim] = 0;
245 }
246 }
247
248 return towerIndex2D;
249 }

375 bool cellCanContainOwnedParticles(index_t index1d) const override {
376 // check if the tower is strictly in the halo region

378 return false;

383 if (index2D[i] < _numTowersPerInteractionLength2D[i] or
384 index2D[i] >= _towersPerDim[i] - _numTowersPerInteractionLength2D[i]) {
385 isHaloTower = true;
