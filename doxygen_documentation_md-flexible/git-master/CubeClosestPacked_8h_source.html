208 static double optimizeSpacingForDensity(const std::array<double, 3> &boxMin, const std::array<double, 3> &boxMax,
209 const double targetDensity, const LatticeStructure structure,
210 const bool centered) {
211 const double volume = (boxMax[0] - boxMin[0]) * (boxMax[1] - boxMin[1]) * (boxMax[2] - boxMin[2]);
212
213 // Spacing for a theoretical infinite lattice (s0)
214 const double spacingExact = std::cbrt(std::sqrt(2.0) / targetDensity);
215 const auto targetParticles = static_cast<size_t>(std::round(targetDensity * volume));
216
217 const auto countAtExactSpacing = calculateParticleCount(boxMin, boxMax, spacingExact, structure, centered);
218
219 if (countAtExactSpacing <= targetParticles) {
220 return spacingExact;
221 }
222
223 // If countAtExactSpacing > targetParticles. We search for a spacing s >= s0 to reduce particle count to
224 // targetParticles.
225 double spacingLow = spacingExact;
226 double spacingHigh = spacingExact * 1.2;
227 size_t countAtHighSpacing = countAtExactSpacing;
228 while (countAtHighSpacing > targetParticles) {
229 countAtHighSpacing = calculateParticleCount(boxMin, boxMax, spacingHigh, structure, centered);
230 spacingLow = spacingHigh;
231 spacingHigh *= 1.2;
232 if (countAtHighSpacing <= 1) {
233 break;
234 }
235 }
236
237 // Binary search for the transition around targetParticles
238 constexpr size_t maxIterations = 20;
239 constexpr double tolerance = 1e-10;
240 for (size_t iter = 0; iter < maxIterations and (spacingHigh - spacingLow) > tolerance; ++iter) {
241 const double spacingMid = spacingLow + 0.5 * (spacingHigh - spacingLow);
242 const size_t countMid = calculateParticleCount(boxMin, boxMax, spacingMid, structure, centered);
243 if (countMid > targetParticles) {
244 spacingLow = spacingMid;
245 } else {
246 spacingHigh = spacingMid;
247 }
248 }
249
250 // spacingLow gives count > targetParticles, spacingHigh gives count <= targetParticles.
251 const size_t countLow = calculateParticleCount(boxMin, boxMax, spacingLow, structure, centered);
252 const size_t countHigh = calculateParticleCount(boxMin, boxMax, spacingHigh, structure, centered);
253
254 const size_t diffLow = (countLow >= targetParticles) ? (countLow - targetParticles) : (targetParticles - countLow);
255 const size_t diffHigh =
256 (countHigh >= targetParticles) ? (countHigh - targetParticles) : (targetParticles - countHigh);
257
258 // Pick whichever is closer to targetParticles
259 if (diffHigh <= diffLow) {
260 return spacingHigh;
261 }
262 return spacingLow;
263 }
