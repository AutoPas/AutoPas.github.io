83template <typename T>
84T parseComplexTypeValueSingle(const YAML::Node &node, const std::string &key,
85 std::vector<std::string> &complexTypeErrors, const bool required = true) {
86 T value;
87 try {
88 value = node[key].as<T>();
89 } catch (const std::exception &e) {
90 if (required) {
91 std::stringstream ss;
92 ss << "Error parsing " << key << ". Make sure that key \"" << key
93 << "\" exists and has the expected value: " << typeToStr<T>();
94 complexTypeErrors.push_back(ss.str());
95 } else {
96 return T{};
97 }
98 }
99 return value;
100}

109template <typename T, size_t S>
110std::array<T, S> parseComplexTypeValueSequence(const YAML::Node &node, const std::string &key,
111 std::vector<std::string> &complexTypeErrors) {
112 std::array<T, S> value;
113 try {
114 YAML::Node n = node[key];
115 for (int i = 0; i < S; i++) {
116 value[i] = n[i].as<T>();
117 }
118
119 } catch (const std::exception &e) {
120 std::stringstream ss;
121 ss << "Error parsing " << key << ". Make sure that key \"" << key << "\" exists and has the expected value: "
122 << "YAML-sequence of " << std::to_string(S) << " " << typeToStr<T>() << " values.";
123 complexTypeErrors.push_back(ss.str());
124 }
125 return value;
126}

135template <typename T>
136std::vector<T> parseComplexTypeValueSequence(const YAML::Node &node, const std::string &key,
137 std::vector<std::string> &complexTypeErrors) {
138 std::vector<T> value;
139 try {
140 YAML::Node n = node[key];
141 const auto vecLength = n.size();
142 value.reserve(vecLength);
143 for (int i = 0; i < vecLength; i++) {
144 value.emplace_back(n[i].as<T>());
145 }
146
147 } catch (const std::exception &e) {
148 std::stringstream ss;
149 ss << "Error parsing " << key << ". Make sure that key \"" << key << "\" exists and has the expected value: "
150 << "YAML-sequence of " << typeToStr<T>() << " values.";
151 complexTypeErrors.push_back(ss.str());
152 }
153 return value;
154}
